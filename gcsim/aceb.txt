fn WaitForCryo() {
  let StartFrame = f();
  if !.element.t0.pyro {      # If the enemy aura is not Pyro, wait for Cryo aura (so if its already Pyro you don't wait)
    while !.element.t0.cryo && f() - StartFrame < 120 {   # wait but up to 120f(2s), so you don't wait forever
      wait(1);
    }
  }
}

    arle charge;

  arle attack:2;
  if .arle.cons < 2 {
    arle attack; # if Arle is Pre-C2, One more NA to make E off-cd
  }
}
